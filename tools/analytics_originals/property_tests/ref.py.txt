"""활동 분석 C1~C7 참조 계산 프로토타입(명세 검증용, 저장소 코드 아님).

입력은 이미 projection된 구간 목록이다. 하루(또는 임의 창) 전체를 한 번에 계산한다.
"""

from __future__ import annotations

from collections import Counter, defaultdict
from dataclasses import dataclass, field
from datetime import datetime, timedelta

UNCLASSIFIED = "__unclassified__"


@dataclass(frozen=True)
class Seg:
    start: datetime
    end: datetime | None  # None: 열린 구간
    last_obs: datetime | None = None  # 활동 구간의 마지막 관찰 시각
    gap: bool = False  # 수집 공백
    opaque: bool = False
    app: str | None = None  # bundleId(captured일 때만)
    browser: bool = False  # web.kind == browser
    site: str | None = None  # URL captured일 때 eTLD+1
    title: str | None = None
    url: str | None = None
    label: str | None = None  # 유효한 확정 라벨(UNCLASSIFIED 포함), 없으면 None
    confirmed_version: str | None = None
    version: str = "v1"

    @property
    def eff_end(self) -> datetime:
        if self.end is not None:
            return self.end
        assert self.last_obs is not None
        return self.last_obs

    @property
    def target(self) -> str | None:
        """사용 대상. 공백·불투명·앱 미확인은 None."""
        if self.gap or self.opaque or self.app is None:
            return None
        if self.browser and self.site is not None:
            return self.site
        return self.app

    @property
    def valid_label(self) -> str | None:
        if self.gap or self.opaque or self.label is None:
            return None
        if self.confirmed_version is not None and self.confirmed_version != self.version:
            return None
        return self.label


def _ov(a0: datetime, a1: datetime, b0: datetime, b1: datetime) -> timedelta:
    return max(timedelta(0), min(a1, b1) - max(a0, b0))


# ---------------- C1 ----------------
@dataclass
class C1:
    apps: defaultdict = field(default_factory=lambda: defaultdict(timedelta))
    targets: defaultdict = field(default_factory=lambda: defaultdict(timedelta))
    unknown_app: timedelta = timedelta(0)
    opaque: timedelta = timedelta(0)
    observed: timedelta = timedelta(0)


def c1(segs: list[Seg], w0: datetime, w1: datetime) -> C1:
    r = C1()
    for s in segs:
        if s.gap:
            continue
        d = _ov(s.start, s.eff_end, w0, w1)
        if not d:
            continue
        r.observed += d
        if s.opaque:
            r.opaque += d
        elif s.app is None:
            r.unknown_app += d
        else:
            r.apps[s.app] += d
            r.targets[s.target] += d
    return r


# ---------------- 전환 공통 ----------------
@dataclass(frozen=True)
class Transition:
    at: datetime
    prev: str
    next: str
    dwell: timedelta
    # 2026-11-13: 뒤 구간의 구간 순서상 위치(C6 이탈 중 전환을 구간 순서로 가른다). 입력 목록에 따라
    # 달라지는 위치라 전환의 같음 비교에서 뺀다(날짜 읽기 결과와 전체 계산 비교).
    seg_index: int = field(default=-1, compare=False)


def _adjacent_pairs(segs: list[Seg]):
    for a, b in zip(segs, segs[1:]):
        if a.gap or b.gap or a.end is None or a.end != b.start:
            continue  # 공백·60초 초과 침묵(end=last_obs < 다음 start)·열린 구간
        yield a, b


def _run_dwell(segs: list[Seg], i: int) -> timedelta:
    """i번째 구간까지 같은 사용 대상에 맞닿아 연속으로 머문 시간."""
    t = segs[i].target
    total = segs[i].eff_end - segs[i].start
    j = i
    while j > 0:
        a, b = segs[j - 1], segs[j]
        if a.gap or a.end != b.start or a.target != t:
            break
        total += a.eff_end - a.start
        j -= 1
    return total


def c2(segs: list[Seg], *, dwell: str = "run", min_dwell: timedelta | None = None,
       drop_zero: bool = False) -> list[Transition]:
    """사용 대상 전환. dwell='run'(명세, 2026-11-11 D2) | 'segment'(대안)."""
    if drop_zero:
        segs = _drop_zero(segs)
    out: list[Transition] = []
    for i, (a, b) in enumerate(zip(segs, segs[1:])):
        if a.gap or b.gap or a.end is None or a.end != b.start:
            continue
        ta, tb = a.target, b.target
        if ta is None or tb is None or ta == tb:
            continue
        d = a.eff_end - a.start if dwell == "segment" else _run_dwell(segs, i)
        out.append(Transition(b.start, ta, tb, d, i + 1))
    if min_dwell is not None:
        out = _drop_short_round_trips(out, min_dwell)
    return out


def _drop_zero(segs: list[Seg]) -> list[Seg]:
    """0초 구간을 없는 것으로 본다(앞 구간이 다음 구간에 맞닿는다)."""
    return [s for s in segs if s.gap or s.end is None or s.end != s.start]


def _drop_short_round_trips(ts: list[Transition], t: timedelta) -> list[Transition]:
    """A→B→A에서 B 체류가 t 미만이고 두 전환이 맞붙어 있으면 둘 다 뺀다."""
    drop: set[int] = set()
    for i in range(len(ts) - 1):
        x, y = ts[i], ts[i + 1]
        if x.next == y.prev and x.prev == y.next and y.at - x.at < t and y.dwell < t:
            drop |= {i, i + 1}
    return [x for i, x in enumerate(ts) if i not in drop]


def in_window(ts, w0, w1):
    return [x for x in ts if w0 <= x.at < w1]


# ---------------- C7 ----------------
@dataclass(frozen=True)
class LabelTransition:
    at: datetime
    prev: str
    next: str


def c7(segs: list[Seg], *, drop_zero: bool = False) -> list[LabelTransition]:
    if drop_zero:
        segs = _drop_zero(segs)
    out = []
    for a, b in _adjacent_pairs(segs):
        la, lb = a.valid_label, b.valid_label
        if la in (None, UNCLASSIFIED) or lb in (None, UNCLASSIFIED) or la == lb:
            continue
        out.append(LabelTransition(b.start, la, lb))
    return out


# ---------------- 세션 ----------------
@dataclass(frozen=True)
class Session:
    start: datetime
    end: datetime
    label: str
    pauses: tuple[tuple[datetime, datetime], ...] | None = ()  # None: 일시중지 정보 없음

    def work_intervals(self) -> list[tuple[datetime, datetime]]:
        out, cur = [], self.start
        for p0, p1 in sorted(self.pauses or ()):
            if p0 > cur:
                out.append((cur, p0))
            cur = max(cur, p1)
        if self.end > cur:
            out.append((cur, self.end))
        return out

    def analysis_time(self) -> timedelta:
        return sum((b - a for a, b in self.work_intervals()), timedelta(0))

    def contains_open(self, t: datetime) -> bool:
        """작업 구간 양 끝을 제외한 열린 구간 판정(C3·세션별 C7)."""
        return any(a < t < b for a, b in self.work_intervals())


def c3(transitions, s: Session):
    return [x for x in transitions if s.contains_open(x.at)]


FOCUS, LAPSE, UNCL, PENDING, OPAQUE = "집중", "이탈", "확정 미분류", "판정 대기", "불투명"


def classify(seg: Seg, s: Session) -> str:
    if seg.opaque:
        return OPAQUE
    lab = seg.valid_label
    if lab is None:
        return PENDING
    if lab == UNCLASSIFIED:
        return UNCL
    return FOCUS if lab == s.label else LAPSE


@dataclass(frozen=True)
class Piece:
    start: datetime
    end: datetime
    seg: Seg
    kind: str


def pieces(segs: list[Seg], s: Session) -> list[Piece]:
    out = []
    for seg in segs:
        if seg.gap:
            continue
        for a, b in s.work_intervals():
            lo, hi = max(seg.start, a), min(seg.eff_end, b)
            if hi > lo or (hi == lo and seg.start == seg.eff_end and a <= lo < b):
                out.append(Piece(lo, hi, seg, classify(seg, s)))
    return out


@dataclass
class C4:
    times: dict
    lapse_count: int
    has_pause_info: bool


@dataclass
class Lapse:
    start: datetime
    end: datetime
    entry: Seg
    prev_focus_target: str | None
    elapsed: timedelta
    lapse_time: timedelta
    transitions: int
    returned: bool


def _elapsed(s: Session, t: datetime) -> timedelta:
    return sum((_ov(a, b, s.start, t) for a, b in s.work_intervals()), timedelta(0))


def lapses(segs: list[Seg], s: Session, transitions) -> list[Lapse]:
    ps = pieces(segs, s)
    pos = {id(x): i for i, x in enumerate(segs)}
    out: list[Lapse] = []
    cur = None
    last_focus_target = None
    for p in ps:
        if p.kind == LAPSE:
            if cur is None:
                cur = dict(start=p.start, entry=p.seg, prev=last_focus_target, time=timedelta(0),
                           entry_i=pos[id(p.seg)])
            cur["time"] += p.end - p.start
        elif p.kind == FOCUS:
            if cur is not None:
                out.append(_close(cur, p.start, True, s, transitions, pos[id(p.seg)]))
                cur = None
            last_focus_target = p.seg.target
    if cur is not None:
        work = s.work_intervals()
        end = work[-1][1] if work else s.end  # 2026-11-13: 세션 분석 시간의 끝(FR-16)
        out.append(_close(cur, end, False, s, transitions, None))
    return out


def _close(cur, end, returned, s, transitions, return_i) -> Lapse:
    """이탈 중 전환(FR-20): C3 전환 중 진입 전환(첫 이탈 구간으로 들어온 전환)과 복귀 전환(복귀 집중
    구간으로 들어간 전환) 사이의 것. 2026-11-13 4단계 A 비교: 시각 (시작, 끝) 대신 구간 순서로 가른다.
    시각으로 가르면 진입·복귀 시각에 놓인 0초 구간의 이탈 중 전환까지 빠진다(STEP4-A-REPORT.md)."""
    inside = [x for x in c3(transitions, s)
              if cur["entry_i"] < x.seg_index and (return_i is None or x.seg_index < return_i)]
    return Lapse(cur["start"], end, cur["entry"], cur["prev"], _elapsed(s, cur["start"]),
                 cur["time"], len(inside), returned)


def c4(segs: list[Seg], s: Session) -> C4:
    times = {k: timedelta(0) for k in (FOCUS, LAPSE, UNCL, PENDING, OPAQUE)}
    for p in pieces(segs, s):
        times[p.kind] += p.end - p.start
    times["미관찰"] = s.analysis_time() - sum(times.values(), timedelta(0))
    return C4(times, len(lapses(segs, s, [])), s.pauses is not None)


UNKNOWN_TARGET = "__앱 미확인__"


def c5(segs: list[Seg], s: Session) -> dict[str, tuple[int, timedelta]]:
    """진입 횟수: 집중 이탈을 시작하는 이탈 조각은 늘 1회, 이탈 중에는 직전 조각의 대상과 다르면 1회.

    2026-11-13 D11: 이탈을 시작하는 조각은 직전 조각과 사용 대상이 같아도 진입이다(진입 합 ≥ 이탈 건수).
    직전 조각은 세션 작업 시간 안의 앞 조각이며 불투명 조각은 건너뛴다(D4·D10).
    앱 미확인 이탈은 UNKNOWN_TARGET 항목으로 모은다(D4).
    """
    entries: Counter = Counter()
    time: defaultdict = defaultdict(timedelta)
    prev_key = None
    lapsing = False  # 집중 이탈 중인가(집중 조각이 끝낸다, FR-16)
    for p in pieces(segs, s):
        if p.kind == OPAQUE:
            continue
        key = p.seg.target if p.seg.target is not None else UNKNOWN_TARGET
        if p.kind == LAPSE:
            time[key] += p.end - p.start
            if not lapsing or key != prev_key:
                entries[key] += 1
            lapsing = True
        elif p.kind == FOCUS:
            lapsing = False
        prev_key = key
    return {k: (entries[k], time[k]) for k in time}


# ---------------- 15분 칸 경로(차등 비교용) ----------------
BIN = timedelta(minutes=15)


def bin_floor(t: datetime, size: timedelta | None = None) -> datetime:
    size = size or BIN
    epoch = datetime(1970, 1, 1, tzinfo=t.tzinfo)
    return epoch + ((t - epoch) // size) * size


def bin_segments(segs: list[Seg], b0: datetime, *, inclusive: bool = True,
                 prev: bool = False, run_back: bool = True,
                 size: timedelta | None = None) -> list[Seg]:
    """칸 재계산이 DB에서 읽는 구간.

    inclusive=True : started_at < 칸 끝 AND 끝 >= 칸 시작 (2026-11-11 D1)
    inclusive=False: started_at < 칸 끝 AND 끝 > 칸 시작 (기각된 HANDOFF 조건)
    prev    : 첫 구간의 직전 구간 1개를 더 읽는다(기각된 방식)
    run_back: 첫 구간에서 같은 사용 대상으로 맞닿은 앞 구간들을 거슬러 읽는다(D2 연속 체류)
    """
    b1 = b0 + (size or BIN)
    idx = [i for i, s in enumerate(segs)
           if s.start < b1 and (_end_for_overlap(s) >= b0 if inclusive else _end_for_overlap(s) > b0)]
    if not idx:
        return []
    # 2026-11-13 명세 1차 반영(D1·D2): 조건으로 고른 행 + 거슬러 읽은 행. 연속 범위로 자르지 않는다.
    # 거슬러 읽기는 끝이 있는 첫 행에서 시작한다(끝이 없는 수집 공백은 날짜보다 앞에서 시작해도 골라진다).
    chosen = set(idx)
    closed = [i for i in idx if not (segs[i].gap and segs[i].end is None)]
    first = closed[0] if closed else idx[0]
    if prev:
        chosen.add(max(0, idx[0] - 1))
    if run_back:
        while first > 0:
            a, b = segs[first - 1], segs[first]
            if a.gap or b.gap or a.end != b.start or a.target is None or a.target != b.target:
                break
            first -= 1
            chosen.add(first)
    return [segs[i] for i in sorted(chosen)]


def _end_for_overlap(s: Seg) -> datetime:
    if s.gap:
        return s.end if s.end is not None else datetime.max.replace(tzinfo=s.start.tzinfo)
    return s.eff_end


def binned(fn, segs: list[Seg], w0: datetime, w1: datetime, *, inclusive: bool = True,
           size: timedelta | None = None, **kw):
    """경계(칸 또는 날짜)마다 fn(그 경계의 구간들)으로 전환을 구하고 경계 안의 전환만 합친다."""
    size = size or BIN
    out = []
    b0 = bin_floor(w0, size)
    while b0 < w1:
        local = bin_segments(segs, b0, inclusive=inclusive, size=size, **kw)
        out += [x for x in fn(local) if b0 <= x.at < b0 + size and w0 <= x.at < w1]
        b0 += size
    return out


def binned_c1(segs, w0, w1, *, inclusive: bool) -> C1:
    total = C1()
    b0 = bin_floor(w0)
    while b0 < w1:
        r = c1(bin_segments(segs, b0, inclusive=inclusive), max(b0, w0), min(b0 + BIN, w1))
        for k, v in r.apps.items():
            total.apps[k] += v
        for k, v in r.targets.items():
            total.targets[k] += v
        total.unknown_app += r.unknown_app
        total.opaque += r.opaque
        total.observed += r.observed
        b0 += BIN
    return total
